struct dentry {
  unsigned int d_flags;         → 0x00040000 (DCACHE_LRU_LIST)
  struct lockref d_lockref;     → count == 0
  struct list_head d_lru;       → { .next, .prev }
  struct hlist_bl_node d_hash;  → { .next, .pprev }
}
↓
super_block {
  struct list_lru s_dentry_lru; → { .node, .xa }
}
↓
list_lru {
  struct xarray xa;             → [memcg_id] → mlru
}
↓
list_lru_memcg {
  struct list_lru_one node[];   → node[nid] → l
}
↓
list_lru_one {
  struct list_head list;        → { .next = oldest, .prev = newest }
  spinlock_t lock;
  long nr_items;
}
